-- 创建数据库
CREATE DATABASE IF NOT EXISTS laboratory_management CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE laboratory_management;

create table laboratory
(
lab_id       int auto_increment
primary key,
lab_name     varchar(50)                         not null comment '实验室名称',
location     varchar(100)                        not null,
capacity     int                                 not null,
open_time    time                                not null comment '开放开始时间',
close_time   time                                not null comment '开放结束时间',
created_time timestamp default CURRENT_TIMESTAMP null,
updated_time timestamp default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP,
constraint name
unique (lab_name)
)
comment '实验室表';

create table equipment
(
equipment_id   int auto_increment
primary key,
lab_id         int                                 not null,
equipment_name varchar(100)                        not null comment '设备名称',
model          varchar(50)                         null,
status         tinyint   default 1                 null comment '状态：0-故障 1-正常',
created_time   timestamp default CURRENT_TIMESTAMP null,
updated_time   timestamp default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP,
constraint equipment_ibfk_1
foreign key (lab_id) references laboratory (lab_id)
on delete cascade
)
comment '设备表';

create index idx_lab_id
on equipment (lab_id);

create index idx_name
on equipment (equipment_name);

create index idx_status
on equipment (status);

create index idx_location
on laboratory (location);

create index idx_name
on laboratory (lab_name);

create table user
(
user_id    int auto_increment
primary key,
account    varchar(50)  not null,
password   varchar(100) not null,
permission tinyint      not null comment '权限：0-管理员 1-教师 2-学生',
constraint account
unique (account)
);

create table admin
(
admin_id int auto_increment
primary key,
user_id  int         not null,
name     varchar(50) not null,
constraint user_id
unique (user_id),
constraint admin_ibfk_1
foreign key (user_id) references user (user_id)
on delete cascade
);

create index idx_admin_name
on admin (name);

create index idx_admin_user
on admin (user_id);

create table maintenance
(
maintenance_id int auto_increment
primary key,
equipment_id   int                                 not null,
user_id        int                                 not null,
report_time    datetime                            not null,
notes          text                                not null comment '问题描述',
status         tinyint   default 0                 null comment '状态：0-待处理 1-处理中 2-已修复',
repair_notes   text                                null comment '维修记录',
created_time   timestamp default CURRENT_TIMESTAMP null,
updated_time   timestamp default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP,
constraint maintenance_ibfk_1
foreign key (equipment_id) references equipment (equipment_id)
on delete cascade,
constraint maintenance_ibfk_2
foreign key (user_id) references user (user_id)
on delete cascade
)
comment '故障报修表';

create index idx_equipment_id
on maintenance (equipment_id);

create index idx_report_time
on maintenance (report_time);

create index idx_status
on maintenance (status);

create index idx_user_id
on maintenance (user_id);

create table teacher
(
teacher_id int auto_increment
primary key,
user_id    int         not null,
name       varchar(50) not null,
number     varchar(20) not null,
department varchar(50) not null,
constraint number
unique (number),
constraint teacher_ibfk_1
foreign key (user_id) references user (user_id)
on delete cascade
);

create table student
(
student_id int auto_increment
primary key,
user_id    int         not null,
teacher_id int         not null,
name       varchar(50) not null,
number     varchar(20) not null,
department varchar(50) not null,
major      varchar(50) not null,
constraint number
unique (number),
constraint student_ibfk_1
foreign key (user_id) references user (user_id)
on delete cascade,
constraint student_ibfk_2
foreign key (teacher_id) references teacher (teacher_id)
on delete cascade
);

create table reservation
(
reservation_id int auto_increment
primary key,
student_id     int                                 not null,
lab_id         int                                 not null,
project_name   varchar(100)                        not null,
start_time     datetime                            not null,
end_time       datetime                            not null,
status         tinyint   default 0                 null comment '状态：0-待审核 1-通过 2-拒绝',
created_time   timestamp default CURRENT_TIMESTAMP null,
updated_time   timestamp default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP,
constraint reservation_ibfk_1
foreign key (student_id) references student (student_id)
on delete cascade,
constraint reservation_ibfk_2
foreign key (lab_id) references laboratory (lab_id)
on delete cascade
)
comment '预约表';

create table approval
(
approval_id    int auto_increment
primary key,
reservation_id int                                 not null,
teacher_id     int                                 not null,
approval_time  datetime                            null,
status         tinyint                             not null comment '结果：1-通过 2-拒绝',
notes          text                                null comment '审核意见',
created_time   timestamp default CURRENT_TIMESTAMP null,
updated_time   timestamp default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP,
constraint uk_reservation_id
unique (reservation_id),
constraint approval_ibfk_1
foreign key (reservation_id) references reservation (reservation_id)
on delete cascade,
constraint approval_ibfk_2
foreign key (teacher_id) references teacher (teacher_id)
on delete cascade
)
comment '审核记录表';

create index idx_approval_time
on approval (approval_time);

create index idx_reservation_id
on approval (reservation_id);

create index idx_status
on approval (status);

create index idx_teacher_id
on approval (teacher_id);

create index idx_end_time
on reservation (end_time);

create index idx_lab_id
on reservation (lab_id);

create index idx_start_time
on reservation (start_time);

create index idx_status
on reservation (status);

create index idx_student_id
on reservation (student_id);

create index idx_time_range
on reservation (start_time, end_time);

create index idx_student_number
on student (number);

create index idx_student_teacher
on student (teacher_id);

create index user_id
on student (user_id);

create index idx_teacher_number
on teacher (number);

create index user_id
on teacher (user_id);

create table user_record
(
record_id      int auto_increment
primary key,
reservation_id int                                 not null,
student_id     int                                 not null,
start_time     datetime                            not null,
end_time       datetime                            null,
notes          text                                null comment '使用备注',
created_time   timestamp default CURRENT_TIMESTAMP null,
updated_time   timestamp default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP,
constraint usage_record_ibfk_1
foreign key (reservation_id) references reservation (reservation_id)
on delete cascade,
constraint usage_record_ibfk_2
foreign key (student_id) references student (student_id)
on delete cascade
)
comment '使用记录表';

create index idx_end_time
on usage_record (end_time);

create index idx_reservation_id
on usage_record (reservation_id);

create index idx_start_time
on usage_record (start_time);

create index idx_student_id
on usage_record (student_id);

create index idx_user_account
on user (account);
